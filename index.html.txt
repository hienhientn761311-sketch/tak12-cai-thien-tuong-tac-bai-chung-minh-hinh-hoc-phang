<script>
    const wrong   = { 'dz-ao': 0, 'dz-od': 0 };
    const hintMap = { 'dz-ao': 'hint-ao', 'dz-od': 'hint-od' };
    const labels   = { 'dz-ao': 'AO = CO', 'dz-od': 'OD = OB' };
    const expected = { 'dz-ao': 'card-ao', 'dz-od': 'card-od' };
    let doneAO = false, doneOD = false;
 
    function onDrag(e) { e.dataTransfer.setData('text', e.target.id); }
    function onOver(e, id) {
        e.preventDefault();
        const el = document.getElementById(id);
        if (!el.classList.contains('filled')) el.classList.add('drag-over');
    }
    function onLeave(id) { document.getElementById(id).classList.remove('drag-over'); }
 
    function fillZone(zoneId, cardId, label) {
        const zone = document.getElementById(zoneId);
        zone.classList.remove('drag-over', 'available', 'shake');
        zone.classList.add('filled');
        zone.removeAttribute('ondragover');
        zone.removeAttribute('ondragleave');
        zone.removeAttribute('ondrop');
        zone.querySelector('.dz-ph').textContent = label;
        zone.querySelector('.step-tag').classList.add('done');
        document.getElementById(cardId).classList.add('used');
        if (hintMap[zoneId]) document.getElementById(hintMap[zoneId]).classList.remove('show');
    }
 
    function reveal() {
        ['arr-l','arr-m','arr-r'].forEach(id =>
            document.getElementById(id).style.opacity = '1');
        document.getElementById('conclusion').style.display = '';
        document.getElementById('success-banner').classList.add('show');
    }
 
    function onDrop(e, zoneId) {
        e.preventDefault();
        const zone = document.getElementById(zoneId);
        zone.classList.remove('drag-over');
        if (zone.classList.contains('filled')) return;
        const dragged = e.dataTransfer.getData('text');
 
        if (dragged === expected[zoneId]) {
            fillZone(zoneId, dragged, labels[zoneId]);
            if (zoneId === 'dz-ao') doneAO = true;
            if (zoneId === 'dz-od') doneOD = true;
            if (doneAO && doneOD) reveal();
        } else {
            wrong[zoneId] = (wrong[zoneId] || 0) + 1;
            zone.classList.remove('shake');
            void zone.offsetWidth;
            zone.classList.add('shake');
            if (wrong[zoneId] >= 1)
                document.getElementById(hintMap[zoneId]).classList.add('show');
        }
    }
</script>
